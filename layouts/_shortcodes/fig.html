Hugo's PNG re-encode is lossless. */ -}} {{- $src := .Get "src" -}} {{- if not $src }}{{ errorf "fig: src required at %s" .Position }}{{ end -}} {{- $r := .Page.Resources.GetMatch $src -}} {{- if not $r }}{{ errorf "fig: no resource %q in %s" $src .Page.File.Filename }}{{ end -}} {{- $alt := .Get "alt" | default "" -}} {{- $cap := .Get "caption" | default "" -}} {{- /* Cap the intrinsic width; these are small diagrams and upscaling a 683px crop to 1400 only makes the file bigger, not the lines sharper. */ -}} {{- $w := cond (gt $r.Width 900) 900 $r.Width -}} {{- $lg := $r.Resize (printf "%dx png" $w) -}} {{- $sm := $r.Resize (printf "%dx png" (div $w 2)) -}}
